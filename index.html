Google Apps Script Code (Copy & Paste in Extensions > Apps Script): 
function doGet(e) {
  if (e && e.parameter && (e.parameter.patientName || e.parameter.date)) {
    return doPost(e);
  }
  return ContentService.createTextOutput("Prescription Studio Webhook is Active and Ready!").setMimeType(ContentService.MimeType.TEXT);
}

function doPost(e) {
  try {
    var ss;
    try {
      ss = SpreadsheetApp.getActiveSpreadsheet();
    } catch(err) {}
    if (!ss) {
      ss = SpreadsheetApp.openById("1XA6S97SodEjr9MdOXryYdiMFZ2iYJ9LXp-P5YgojGNY");
    }
    var sheet = ss.getSheets()[0];
    
    var data = {};
    if (e && e.postData && e.postData.contents) {
      try {
        data = JSON.parse(e.postData.contents);
      } catch(err) {
        data = e.parameter || {};
      }
    } else if (e && e.parameter) {
      data = e.parameter;
    }

    if (sheet.getLastRow() === 0) {
      sheet.appendRow(["Timestamp", "Prescription Date", "Patient Name", "Age/Sex", "Phone", "Address/City", "Drug Allergies", "Chief Complaints", "Diagnosis", "Medications Summary", "General Advice", "Investigations", "Follow-up", "Doctor Name"]);
    }
    sheet.appendRow([
      new Date(),
      data.date || "",
      data.patientName || "",
      data.ageSex || "",
      data.phone || "",
      data.address || "",
      data.drugAllergies || "None",
      data.chiefComplaints || "",
      data.diagnosis || "",
      data.medicationsSummary || "",
      data.advice || "",
      data.investigations || "",
      data.followUp || "",
      data.doctorName || ""
    ]);
    return ContentService.createTextOutput(JSON.stringify({status: "success"})).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({status: "error", message: err.toString()})).setMimeType(ContentService.MimeType.JSON);
  }
}
